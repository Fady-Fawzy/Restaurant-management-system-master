cmake_minimum_required(VERSION 3.16)
project(RestaurantSimulation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/Restaurant-management-system-master/Restaurant management system")
file(GLOB RESTAURANT_SOURCES CONFIGURE_DEPENDS "${SOURCE_DIR}/*.cpp")
add_executable(restaurant ${RESTAURANT_SOURCES})
include(CTest)
if(BUILD_TESTING)
  add_executable(data_structures_test tests/data_structures.cpp)
  target_include_directories(data_structures_test PRIVATE "${SOURCE_DIR}")
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(data_structures_test PRIVATE -Wall -Wextra -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(data_structures_test PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME data_structures COMMAND data_structures_test)
endif()
